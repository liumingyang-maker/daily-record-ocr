# Phase 5: MiMo API 接入 Implementation Plan

**Goal:** 实现 MiMo 视觉大模型 API 集成，支持 record_level 模式对记录图片进行复核识别。第一版使用 mock API 调用，后续替换为真实 MiMo API。

**Architecture:** Infrastructure 层的 vision 模块封装 MiMo API 调用。Application 层编排 MiMo 识别流程。支持缓存、日志、超时重试。

## Global Constraints

- MiMo 独立模块，可关闭（S13.3）
- API Key 从 .env 读取（MIMO_API_KEY）
- 支持 timeout、retry、cache
- 失败不中断任务，进入人工确认
- 每条记录裁剪图调用一次 MiMo
- 请求日志写入 mimo_request_logs 表
- 缓存写入 mimo_cache 表

```python
# app/application/mimo_service.py
from app.infrastructure.database.session import get_session
from app.infrastructure.database.models import (
    RecognitionJob, ProductionRecord, FieldRecognitionResult,
    MimoRequestLog, MimoCache,
)
from app.infrastructure.vision import get_mimo_client
from app.infrastructure.image.cropper import crop_field_cells
from app.configs import load_config
import cv2
import hashlib
import time


class MimoService:
    def __init__(self):
        self.template = load_config("template_daily_record_v1")["template"]
        self.client = get_mimo_client()
        self.config = load_config("mimo").get("mimo", {})

    def recognize_job(self, job_id: int) -> dict:
        if not self.config.get("enabled", True):
            return {"job_id": job_id, "skipped": True, "reason": "MiMo disabled"}

        with get_session() as session:
            job = session.query(RecognitionJob).get(job_id)
            if not job:
                raise ValueError(f"Job not found: {job_id}")
            records = session.query(ProductionRecord).filter_by(job_id=job_id).all()
            record_data = [(r.id, r.record_index, r.record_crop_path) for r in records]

        results = []
        for record_id, record_index, crop_path in record_data:
            if not crop_path:
                continue
            img = cv2.imread(crop_path)
            if img is None:
                continue

            # 检查缓存
            image_hash = self._hash_file(crop_path)
            cached = self._check_cache(image_hash)
            if cached:
                mimo_result = cached
            else:
                # 调用 MiMo
                start = time.time()
                try:
                    mimo_result = self.client.recognize_record(crop_path)
                    latency = int((time.time() - start) * 1000)
                    self._log_request(job_id, record_id, image_hash, mimo_result, latency)
                    if mimo_result.success:
                        self._save_cache(image_hash, mimo_result)
                except Exception as e:
                    mimo_result = None
                    self._log_request(job_id, record_id, image_hash, None, 0, str(e))

            if not mimo_result or not mimo_result.success:
                continue

            # 更新字段识别结果
            for field_key, field_data in mimo_result.fields.items():
                with get_session() as session:
                    existing = session.query(FieldRecognitionResult).filter_by(
                        job_id=job_id, record_id=record_id, field_key=field_key
                    ).first()
                    if existing:
                        existing.mimo_raw_text = field_data.get("raw_text", "")
                        existing.mimo_confidence = field_data.get("confidence", 0)
                        # 如果 MiMo 置信度更高，更新最终值
                        if field_data.get("confidence", 0) > (existing.ocr_confidence or 0):
                            existing.final_value = field_data["value"]
                            existing.final_confidence = field_data["confidence"]
                            existing.source = "mimo"
                    else:
                        fr = FieldRecognitionResult(
                            job_id=job_id,
                            record_id=record_id,
                            field_key=field_key,
                            mimo_raw_text=field_data.get("raw_text", ""),
                            mimo_confidence=field_data.get("confidence", 0),
                            final_value=field_data["value"],
                            final_confidence=field_data["confidence"],
                            source="mimo",
                            need_review=field_data.get("confidence", 0) < 0.95,
                        )
                        session.add(fr)

            results.append({
                "record_id": record_id,
                "fields_count": len(mimo_result.fields),
            })

        # 更新 job
        with get_session() as session:
            job = session.query(RecognitionJob).get(job_id)
            job.mimo_used = True
            job.mimo_mode = "record_level"

        return {"job_id": job_id, "records_processed": len(results), "results": results}

    def _hash_file(self, path: str) -> str:
        h = hashlib.md5()
        with open(path, "rb") as f:
            h.update(f.read(8192))
        return h.hexdigest()

    def _check_cache(self, image_hash: str):
        from app.infrastructure.vision.mimo_client import MimoResult
        with get_session() as session:
            cache = session.query(MimoCache).filter_by(image_hash=image_hash).first()
            if cache and cache.response_json:
                return MimoResult(**cache.response_json)
        return None

    def _save_cache(self, image_hash: str, result):
        with get_session() as session:
            cache = MimoCache(
                image_hash=image_hash,
                prompt_version=self.config.get("prompt_version", "v1"),
                model=self.config.get("model", "mock"),
                response_json={
                    "fields": result.fields,
                    "raw_response": result.raw_response,
                    "success": result.success,
                },
            )
            session.add(cache)

    def _log_request(self, job_id, record_id, image_hash, result, latency, error=""):
        with get_session() as session:
            log = MimoRequestLog(
                job_id=job_id,
                record_id=record_id,
                request_type="record_level",
                model=self.config.get("model", "mock"),
                image_hash=image_hash,
                success=result.success if result else False,
                error_message=error or (result.error if result else ""),
                latency_ms=latency,
                response_json={"fields": result.fields} if result and result.success else None,
            )
            session.add(log)
```
