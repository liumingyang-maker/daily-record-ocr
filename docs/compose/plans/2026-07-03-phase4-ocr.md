# Phase 4: PP-OCR 单格识别 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use compose:subagent (recommended) or compose:execute to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 OCR 识别流程：对切好的字段小格子图片调用 OCR 引擎识别文字，将识别结果写入 field_recognition_results 表。第一版使用 mock OCR 引擎（可替换为 PaddleOCR）。

**Architecture:** Infrastructure 层的 ocr 模块提供 OCR 引擎抽象接口，第一版实现 mock 引擎。Application 层的 ocr_service 编排识别流程。Interface 层提供触发识别的 API。OCR 引擎可替换（架构约束 S13.2）。

**Tech Stack:** Python, SQLAlchemy, 依赖 Phase 3 的 cell crop 图片

## Global Constraints

- OCR 引擎抽象成接口，可替换（S13.2）
- 每个字段保存多来源结果（S13.6）
- 不确定字段必须进入人工确认（S8）
- 第一版使用 mock 引擎，返回固定测试数据
- 字段识别结果写入 field_recognition_results 表
- 每个字段同时保存 ocr_raw_text 和 ocr_confidence

from app.infrastructure.ocr.engine import OCREngine, OCRResult


class MockEngine(OCREngine):
    """模拟 OCR 引擎，返回预设测试数据"""

    # 预设识别结果，key 为图片路径中的特征字符串
    PRESET = {
        "time": OCRResult(text="08:30", confidence=0.95),
        "customer_queue": OCRResult(text="客户A-01", confidence=0.92),
        "color": OCRResult(text="红色", confidence=0.88),
        "date_batch_no": OCRResult(text="2026-07-03-B001", confidence=0.90),
        "material": OCRResult(text="PP原料", confidence=0.85),
        "usage": OCRResult(text="100", confidence=0.93),
        "main_speed": OCRResult(text="1500", confidence=0.91),
        "temperature": OCRResult(text="180", confidence=0.89),
    }

    def recognize(self, image_path: str) -> OCRResult:
        for key, result in self.PRESET.items():
            if key in image_path:
                return result
        return OCRResult(text="[未识别]", confidence=0.0)

```python
# app/application/ocr_service.py
from app.infrastructure.database.session import get_session
from app.infrastructure.database.models import (
    RecognitionJob, ProductionRecord, FieldRecognitionResult
)
from app.infrastructure.ocr import get_ocr_engine
from app.infrastructure.image.cropper import crop_field_cells
from app.configs import load_config
import cv2


class OcrService:
    def __init__(self):
        self.template = load_config("template_daily_record_v1")["template"]
        self.engine = get_ocr_engine()

    def recognize_job(self, job_id: int) -> dict:
        """对一个 job 的所有记录执行 OCR 识别"""
        with get_session() as session:
            job = session.query(RecognitionJob).get(job_id)
            if not job:
                raise ValueError(f"Job not found: {job_id}")
            if job.status != "PREPROCESSED":
                raise ValueError(f"Job status must be PREPROCESSED, got: {job.status}")
            job.status = "RECOGNIZING"
            session.flush()

        # 获取记录
        with get_session() as session:
            records = session.query(ProductionRecord).filter_by(job_id=job_id).all()
            record_data = [(r.id, r.record_index, r.record_crop_path) for r in records]

        results = []
        for record_id, record_index, crop_path in record_data:
            if not crop_path:
                continue
            img = cv2.imread(crop_path)
            if img is None:
                continue

            # 切割字段
            cells = crop_field_cells(img, self.template, str(job_id), record_index)

            # 逐字段 OCR
            for field_name, cell_path in cells.items():
                ocr_result = self.engine.recognize(cell_path)
                with get_session() as session:
                    fr = FieldRecognitionResult(
                        job_id=job_id,
                        record_id=record_id,
                        field_key=field_name,
                        field_label=field_name,
                        cell_crop_path=cell_path,
                        ocr_raw_text=ocr_result.text,
                        ocr_confidence=ocr_result.confidence,
                        final_value=ocr_result.text,
                        final_confidence=ocr_result.confidence,
                        source="ocr",
                        need_review=ocr_result.confidence < 0.95,
                    )
                    session.add(fr)
                results.append({
                    "record_id": record_id,
                    "field": field_name,
                    "text": ocr_result.text,
                    "confidence": ocr_result.confidence,
                })

        # 更新 job 状态
        with get_session() as session:
            job = session.query(RecognitionJob).get(job_id)
            job.status = "NEED_REVIEW"

        return {
            "job_id": job_id,
            "recognized_fields": len(results),
            "results": results,
        }
```
